package ppfootballmanager.utils;

import ppfootballmanager.models.*;
import ppfootballmanager.services.MatchSimulatorStrategyImpl;

import com.ppstudios.footballmanager.api.contracts.league.IStanding;
import com.ppstudios.footballmanager.api.contracts.match.IMatch;
import com.ppstudios.footballmanager.api.contracts.team.IClub;
import com.ppstudios.footballmanager.api.contracts.team.ITeam;
import com.ppstudios.footballmanager.api.contracts.player.IPlayer;

import java.util.Scanner;

/**
 * Handles the user interaction with the game
 */
public class Menu {
    private SeasonImpl season;
    private Scanner scanner;
    private boolean running;

    /**
     * Constructor for the Menu class
     * 
     * @param season The season to manage
     */
    public Menu(SeasonImpl season) {
        this.season = season;
        this.scanner = new Scanner(System.in);
        this.running = true;
        
        // Set the match simulator strategy
        season.setMatchSimulator(new MatchSimulatorStrategyImpl());
    }

    /**
     * Starts the menu system
     */
    public void start() {
        while (running) {
            displayMainMenu();
            int choice = getUserChoice();
            handleMainMenuChoice(choice);
        }
        
        scanner.close();
    }

    /**
     * Displays the main menu options
     */
    private void displayMainMenu() {
        System.out.println("\n=== PP Football Manager ===");
        System.out.println("1. View Team Information");
        System.out.println("2. View League Schedule");
        System.out.println("3. View League Standings");
        System.out.println("4. Simulate Next Round");
        System.out.println("5. Simulate Full Season");
        System.out.println("6. Export Results to JSON");
        System.out.println("0. Exit");
        System.out.print("Enter your choice: ");
    }

    /**
     * Gets the user's menu choice
     * 
     * @return The selected menu option
     */
    private int getUserChoice() {
        try {
            return Integer.parseInt(scanner.nextLine());
        } catch (NumberFormatException e) {
            return -1; // Invalid choice
        }
    }

    /**
     * Handles the selected main menu option
     * 
     * @param choice The selected menu option
     */
    private void handleMainMenuChoice(int choice) {
        switch (choice) {
            case 1:
                viewTeamInformation();
                break;
            case 2:
                viewLeagueSchedule();
                break;
            case 3:
                viewLeagueStandings();
                break;
            case 4:
                simulateNextRound();
                break;
            case 5:
                simulateFullSeason();
                break;
            case 6:
                exportToJson();
                break;
            case 0:
                running = false;
                System.out.println("Thank you for playing PP Football Manager!");
                break;
            default:
                System.out.println("Invalid choice. Please try again.");
        }
    }

    /**
     * Displays information about the teams
     */
    private void viewTeamInformation() {
        System.out.println("\n=== Team Information ===");
        
        // Get all clubs in the season
        IClub[] clubs = season.getCurrentClubs();
        if (clubs == null || clubs.length == 0) {
            System.out.println("No teams found in the league.");
            waitForEnter();
            return;
        }
        
        // Display list of teams
        for (int i = 0; i < clubs.length; i++) {
            System.out.println((i + 1) + ". " + clubs[i].getName());
        }
        
        System.out.println("0. Back to Main Menu");
        System.out.print("Select a team to view details: ");
        
        try {
            int teamChoice = Integer.parseInt(scanner.nextLine());
            if (teamChoice == 0) return;
            
            if (teamChoice > 0 && teamChoice <= clubs.length) {
                displayTeamDetails(clubs[teamChoice - 1]);
            } else {
                System.out.println("Invalid team selection.");
            }
        } catch (NumberFormatException e) {
            System.out.println("Invalid input. Please enter a number.");
        }
        
        waitForEnter();
    }
    
    /**
     * Displays detailed information about a specific team
     * 
     * @param club The club to display details for
     */
    private void displayTeamDetails(IClub club) {
        System.out.println("\n=== " + club.getName() + " (" + club.getCode() + ") ===");
        
        // Find team in the season
        ITeam team = null;
        for (IMatch match : season.getMatches()) {
            if (match.getHomeTeam().getClub().equals(club)) {
                team = match.getHomeTeam();
                break;
            } else if (match.getAwayTeam().getClub().equals(club)) {
                team = match.getAwayTeam();
                break;
            }
        }
        
        if (team == null) {
            System.out.println("Team information not available.");
            return;
        }
        
        // Display team stats
        IStanding[] standings = season.getLeagueStandings();
        for (IStanding standing : standings) {
            if (standing instanceof StandingImpl) {
                StandingImpl standingImpl = (StandingImpl) standing;
                if (standingImpl.getClub().equals(club)) {
                    System.out.println("\nCurrent Position: " + standingImpl.getPosition());
                    System.out.println("Played: " + standingImpl.getPlayed());
                    System.out.println("Won: " + standingImpl.getWon());
                    System.out.println("Drawn: " + standingImpl.getDrawn());
                    System.out.println("Lost: " + standingImpl.getLost());
                    System.out.println("Goals For: " + standingImpl.getGoalsFor());
                    System.out.println("Goals Against: " + standingImpl.getGoalsAgainst());
                    System.out.println("Goal Difference: " + standingImpl.getGoalDifference());
                    System.out.println("Points: " + standingImpl.getPoints());
                    break;
                }
            }
        }
        
        // Display squad information
        IPlayer[] players = team.getPlayers();
        if (players != null && players.length > 0) {
            System.out.println("\nSquad:");
            System.out.printf("%-4s %-25s %-10s %-12s\n", "No.", "Name", "Position", "Nationality");
            System.out.println("------------------------------------------------------");
            
             for (IPlayer player : players) {
            System.out.printf("%-4d %-25s %-10s %-12s\n", 
                player.getNumber(), 
                player.getName(), 
                player.getPosition().toString(), // Changed from getPositionCode() to toString()
                player.getNationality());
        }
        } else {
            System.out.println("\nNo player information available.");
        }
        
        // Display fixtures for this team
        IMatch[] teamMatches = season.getSchedule().getMatchesForTeam(team);
        if (teamMatches != null && teamMatches.length > 0) {
            System.out.println("\nFixtures:");
            System.out.println("------------------------------------------------------");
            
            for (IMatch match : teamMatches) {
                String result = !match.isPlayed() ? "Not played" : season.displayMatchResult(match);
                System.out.printf("Round %-2d: %s\n", match.getRound(), result);
            }
        }
    }

    /**
     * Displays the league schedule
     */
    private void viewLeagueSchedule() {
        System.out.println("\n=== League Schedule ===");
        
        int maxRounds = season.getMaxRounds();
        if (maxRounds == 0) {
            System.out.println("Schedule not available.");
            waitForEnter();
            return;
        }
        
        System.out.println("Select a round to view (1-" + maxRounds + "), or 0 for all rounds: ");
        try {
            int roundChoice = Integer.parseInt(scanner.nextLine());
            
            if (roundChoice == 0) {
                // Display all rounds
                for (int round = 1; round <= maxRounds; round++) {
                    displayRoundMatches(round);
                }
            } else if (roundChoice >= 1 && roundChoice <= maxRounds) {
                // Display specific round
                displayRoundMatches(roundChoice);
            } else {
                System.out.println("Invalid round selection.");
            }
        } catch (NumberFormatException e) {
            System.out.println("Invalid input. Please enter a number.");
        }
        
        waitForEnter();
    }
    
    /**
     * Displays matches for a specific round
     * 
     * @param round The round number
     */
    private void displayRoundMatches(int round) {
        IMatch[] matches = season.getMatches(round);
        
        System.out.println("\nRound " + round + ":");
        System.out.println("------------------------------------------------------");
        
        if (matches == null || matches.length == 0) {
            System.out.println("No matches found for this round.");
            return;
        }
        
        for (IMatch match : matches) {
            if (match.isPlayed()) {
                String result = season.displayMatchResult(match);
                System.out.println(result);
            } else {
                System.out.println(match.getHomeTeam().getClub().getName() + " vs " + 
                                   match.getAwayTeam().getClub().getName() + " (Not played)");
            }
        }
    }

    /**
     * Displays the current league standings
     */
    private void viewLeagueStandings() {
        System.out.println("\n=== League Standings ===");
        
        IStanding[] standings = season.getLeagueStandings();
        
        if (standings == null || standings.length == 0) {
            System.out.println("No standings available.");
            waitForEnter();
            return;
        }
        
        // Display header
        System.out.printf("%-3s %-20s %-3s %-3s %-3s %-3s %-7s %-3s\n", 
                "Pos", "Club", "Pld", "W", "D", "L", "GF-GA", "Pts");
        System.out.println("----------------------------------------------------------");
        
        // Display standings
        for (IStanding standing : standings) {
            if (standing instanceof StandingImpl) {
                StandingImpl s = (StandingImpl) standing;
                System.out.printf("%-3d %-20s %-3d %-3d %-3d %-3d %-3d-%-3d %-3d\n", 
                        s.getPosition(),
                        s.getClub().getName(),
                        s.getPlayed(),
                        s.getWon(),
                        s.getDrawn(),
                        s.getLost(),
                        s.getGoalsFor(),
                        s.getGoalsAgainst(),
                        s.getPoints());
            }
        }
        
        waitForEnter();
    }

    /**
     * Simulates the next round of matches
     */
    private void simulateNextRound() {
        System.out.println("\n=== Simulating Next Round ===");
        
        if (season.isSeasonComplete()) {
            System.out.println("The season is already complete.");
            waitForEnter();
            return;
        }
        
        int roundToSimulate = season.getCurrentRound() + 1;
        System.out.println("Simulating Round " + roundToSimulate + "...");
        
        // Simulate next round
        season.simulateRound();
        
        // Display results
        displayRoundMatches(roundToSimulate);
        
        waitForEnter();
    }

    /**
     * Simulates the entire season
     */
    private void simulateFullSeason() {
        System.out.println("\n=== Simulating Full Season ===");
        
        if (season.isSeasonComplete()) {
            System.out.println("The season is already complete.");
            waitForEnter();
            return;
        }
        
        System.out.println("Simulating all remaining rounds...");
        
        // Simulate all remaining rounds
        season.simulateSeason();
        
        System.out.println("Season simulation complete!");
        
        // Display final standings
        viewLeagueStandings();
    }
    
    /**
     * Exports season data to JSON
     */
    private void exportToJson() {
        System.out.println("\n=== Exporting to JSON ===");
        
        try {
            season.exportToJson();
            System.out.println("Season data exported to JSON successfully!");
        } catch (Exception e) {
            System.out.println("Error exporting to JSON: " + e.getMessage());
        }
        
        waitForEnter();
    }
    
    /**
     * Waits for the user to press Enter
     */
    private void waitForEnter() {
        System.out.println("\nPress Enter to continue...");
        scanner.nextLine();
    }
}
